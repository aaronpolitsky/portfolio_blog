nats_east_2023_run1 <-
  nats_east_2023_run1 %>%
  select(`GPS_TIME (ms)`:`LAT (deg)`, 
         `SPEED (m/s)`,
         `ACCEL_X (m/s^2)`:`STEERING (deg)`,
         #ANALOG_1:ANALOG_4,
         MATH_HEADING, 
         MATH_ACCEL_2D_SUM,
         MATH_USER_DEFINED_1) %>%
  arrange(`GPS_TIME (ms)`) %>%
  rename(lat = `LAT (deg)`, 
         long = `LONG (deg)`,
         rear_wheelspeed_minus_front = MATH_USER_DEFINED_1) %>%
  mutate(across(
    c(lat, long),
    .fns = \(x) lead(x, n = 5, order_by = `GPS_TIME (ms)`) - x,
    .names = "{col}_delta_200ms"
  )) %>% 
  select(-c(lat, long)) %>% 
  filter(`GPS_TIME (ms)` < 1694012335000) # in the finish chute of the run

lat_mod <- 
  rand_forest(mtry = tune(), min_n = tune(), trees = 1000) %>% 
  set_engine("ranger", 
             num.threads = cores,
             importance = "impurity") %>% 
  set_mode("regression")


lat_wflow <- 
  workflow() %>% 
  add_model(lat_mod) %>% 
  add_recipe(lat_recipe) 

set.seed(99)

run1_with_preds <- 
  nats_east_2023_run1 %>% 
  bind_cols(fitted_lat %>% predict(new_data = nats_east_2023_run1)) %>%
  rename(pred_lat_delta_200ms = .pred) %>% 
  bind_cols(fitted_long %>% predict(new_data = nats_east_2023_run1)) %>%
  rename(pred_long_delta_200ms = .pred) %>% 
  mutate(distance_error = sqrt((pred_long_delta_200ms - long_delta_200ms)^2 + 
                                 (pred_lat_delta_200ms - lat_delta_200ms)^2)) %>% 
  mutate(lat_error = pred_lat_delta_200ms - lat_delta_200ms,
         long_error = pred_long_delta_200ms - long_delta_200ms) 


run1_with_preds %>% 
  ggplot(aes(x = lat_error, y = long_error)) + geom_density2d()
